/* ========================================================
   Googleログイン成功時に実行される関数
======================================================== */
async function handleCredentialResponse(response) {
    try {
        // Googleから送られてきた暗号化データ（JWT）を解読してユーザー情報を取得
        const responsePayload = parseJwt(response.credential);
        const email = responsePayload.email;
        const name = responsePayload.name;
        const picture = responsePayload.picture; // Googleのアイコン画像URL

        // 🌟【重要】学校のドメインを指定して制限をかける
        // 例: "@taira-th.ed.jp" など、実際の学校のメールアドレスの末尾に書き換えてください
        const allowedDomain = "@fcs.ed.jp"; 
        
        if (!email.endsWith(allowedDomain)) {
            alert("無効なアカウントです。\n学校指定のGoogleアカウント（" + allowedDomain + "）でログインしてください。");
            return;
        }

        // 認証が通ったら、ユーザー情報をブラウザ（localStorage）に安全に保存
        localStorage.setItem("userEmail", email);
        localStorage.setItem("userName", name);
        if (picture) {
            localStorage.setItem("profileImage", picture); // Googleのプロフィール写真をそのままアプリに反映
        }
        
        alert("ログインに成功しました！\nこんにちは、" + name + "さん");
        
        // メイン画面（ホーム）へジャンプ！
        window.location.href = "home.html";

    } catch (error) {
        console.error("ログイン処理中にエラーが発生しました:", error);
        alert("ログイン処理に失敗しました。もう一度お試しください。");
    }
}

/* ========================================================
   JWT（Googleの暗号化データ）を解読するための補助関数
======================================================== */
function parseJwt(token) {
    var base64Url = token.split('.')[1];
    var base64 = base64Url.replace(/-/g, '+').replace(/_/g, '/');
    var jsonPayload = decodeURIComponent(atob(base64).split('').map(function(c) {
        return '%' + ('00' + c.charCodeAt(0).toString(16)).slice(-2);
    }).join(''));
    return JSON.parse(jsonPayload);
}

    }
};

const GAS_URL = "https://script.google.com/macros/s/AKfycbz6psDpbuHyxG-8bAIkCKOb6g930qiJUSYJzKKyYPsCPPzMcKIp-MTeK6sh58hViEXA/exec";

function closeMenu() {

function switchTab(mode) {
    document.getElementById("recruit-form-area").style.display = (mode === 'recruit') ? "block" : "none";
    document.getElementById("search-area").style.display = (mode === 'search') ? "block" : "none";
    document.getElementById("tab-recruit-btn").classList.toggle("active", mode === 'recruit');
    document.getElementById("tab-search-btn").classList.toggle("active", mode === 'search');
    if(mode === 'recruit') {
        document.getElementById("rec-name").value = localStorage.getItem("userName") || "未設定";
    }
}

/* =========================
   募集一覧を読み込む関数（削除ボタン表示に対応）
========================= */
async function loadRecruitments() {
    const myEmail = localStorage.getItem("userEmail");
    const listDiv = document.getElementById("user-list");
    const myPostContentDiv = document.getElementById("my-post-content");

    listDiv.innerHTML = `
        <div class="loading-box">
            <div class="spinner"></div>
            <p>データを更新中...</p>
        </div>
    `;
    myPostContentDiv.innerHTML = "更新中...";

    try {
        const res = await fetch(GAS_URL, {
            method: "POST",
            body: JSON.stringify({ mode: "getRecruitments", email: myEmail })
        });

        const result = await res.json();
        
        listDiv.innerHTML = "";
        myPostContentDiv.innerHTML = "投稿している募集はありません。";

        const items = result.message.reverse();
        const myPosts = items.filter(item => item.email === myEmail);
        const otherPosts = items.filter(item => item.email !== myEmail);

        if (myPosts.length > 0) {
            const myItem = myPosts[0];
            myPostContentDiv.innerHTML = `
                <div style="font-weight: bold; margin-bottom: 5px; color:#1cbdc5;">${myItem.grade} | ${myItem.dept}</div>
                <div style="word-wrap: break-word; line-height: 1.4; margin-bottom: 10px;">${myItem.content}</div>
                <button onclick="deleteMyRecruitment()" style="width: 100%; background: #ff4d4d; color: white; border: none; padding: 5px; border-radius: 4px; cursor: pointer; font-size: 12px; font-weight: bold;">
                    ✖ この募集を削除する
                </button>
            `;
        }

        if (otherPosts.length === 0) {
            listDiv.innerHTML = "<p>現在、ユーザーの募集はありません</p>";
        } else {
            const fragment = document.createDocumentFragment();
            otherPosts.forEach(item => {
                const card = document.createElement("div");
                card.className = "user-card";
                card.innerHTML = `
                    <div class="user-badge">${item.grade} | ${item.dept}</div>
                    <h3>${item.name}</h3>
                    <p>${item.content}</p>
                    <button class="btn-match" onclick="startChat('${item.email}')">チャット</button>
                `;
                fragment.appendChild(card);
            });
            listDiv.appendChild(fragment);
        }

    } catch(e) {
        console.error(e);
        listDiv.innerText = "読み込みエラーが発生しました。";
    }
}

async function deleteMyRecruitment() {
    if (!confirm("本当にこの募集を削除しますか？")) return;
    const myEmail = localStorage.getItem("userEmail");
    try {
        const res = await fetch(GAS_URL, {
            method: "POST",
            body: JSON.stringify({ mode: "deleteRecruitment", email: myEmail })
        });
        const result = await res.json();
        if (result.status === "success") {
            alert("募集を削除しました。");
            loadRecruitments();
        } else {
            alert("削除に失敗しました。");
        }
    } catch (e) {
        alert("通信エラーが発生しました。");
    }
}

async function submitRecruitment() {
    const data = {
        mode: "postRecruitment",
        name: localStorage.getItem("userName"),
        grade: document.getElementById("rec-grade").value,
        dept: document.getElementById("rec-dept").value,
        content: document.getElementById("rec-content").value,
        email: localStorage.getItem("userEmail")
    };
    await fetch(GAS_URL, { method: "POST", body: JSON.stringify(data) });
    alert("投稿しました");
    switchTab('search');
    loadRecruitments();
    document.getElementById("rec-content").value = "";
}

function startChat(email) {
    localStorage.setItem("chatTarget", email);
    location.href = 'messages.html';
}

function logout() {
    localStorage.clear();
    window.location.href = "index.html";
}

window.onload = () => {
    const userName = localStorage.getItem("userName");
    const profileImage = localStorage.getItem("profileImage");
    document.getElementById("mypage-text").innerText = userName || "マイページ";

    if(profileImage){
        document.getElementById("profile-icon").src = profileImage;
    } else {
        document.getElementById("profile-icon").src = "https://cdn-icons-png.flaticon.com/512/149/149071.png";
    }
    loadRecruitments();
};
