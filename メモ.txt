/* ========================================================
   Googleログイン成功時に実行される関数
======================================================== */
async function handleCredentialResponse(response) {
    try {
        // Googleから送られてきた暗号化データ（JWT）を解読してユーザー情報を取得
        const responsePayload = parseJwt(response.credential);
        const email = responsePayload.email;
        const name = responsePayload.name;
        const picture = responsePayload.picture; // Googleのアイコン画像URL

        // 🌟【重要】学校のドメインを指定して制限をかける
        // 例: "@taira-th.ed.jp" など、実際の学校のメールアドレスの末尾に書き換えてください
        const allowedDomain = "@fcs.ed.jp"; 
        
        if (!email.endsWith(allowedDomain)) {
            alert("無効なアカウントです。\n学校指定のGoogleアカウント（" + allowedDomain + "）でログインしてください。");
            return;
        }

        // 認証が通ったら、ユーザー情報をブラウザ（localStorage）に安全に保存
        localStorage.setItem("userEmail", email);
        localStorage.setItem("userName", name);
        if (picture) {
            localStorage.setItem("profileImage", picture); // Googleのプロフィール写真をそのままアプリに反映
        }
        
        alert("ログインに成功しました！\nこんにちは、" + name + "さん");
        
        // メイン画面（ホーム）へジャンプ！
        window.location.href = "home.html";

    } catch (error) {
        console.error("ログイン処理中にエラーが発生しました:", error);
        alert("ログイン処理に失敗しました。もう一度お試しください。");
    }
}

/* ========================================================
   JWT（Googleの暗号化データ）を解読するための補助関数
======================================================== */
function parseJwt(token) {
    var base64Url = token.split('.')[1];
    var base64 = base64Url.replace(/-/g, '+').replace(/_/g, '/');
    var jsonPayload = decodeURIComponent(atob(base64).split('').map(function(c) {
        return '%' + ('00' + c.charCodeAt(0).toString(16)).slice(-2);
    }).join(''));
    return JSON.parse(jsonPayload);
}

// 既にログイン済みの場合は自動でホームへ飛ばす処理
window.onload = () => {
    if (localStorage.getItem("userEmail")) {
        window.location.href = "home.html";
    }
};
</script>
