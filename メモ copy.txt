    <script>
        const GAS_URL = "https://script.google.com/macros/s/AKfycbz6psDpbuHyxG-8bAIkCKOb6g930qiJUSYJzKKyYPsCPPzMcKIp-MTeK6sh58hViEXA/exec";
        const myEmail = localStorage.getItem("userEmail");
        let currentTarget = "";

        window.onload = () => {
            const t = localStorage.getItem("chatTarget");
            if(t) { openChat(t); localStorage.removeItem("chatTarget"); }
            else { showList(); }
            setInterval(loadMessages, 3000);
        };

        function showList() {
            const list = JSON.parse(localStorage.getItem("chatHistory") || "[]");
            const area = document.getElementById("chat-history-list");
            area.innerHTML = list.length ? "" : "<p style='padding:20px;'>トーク履歴はありません</p>";
            list.forEach(email => {
                const div = document.createElement("div");
                div.style.padding = "20px; border-bottom:1px solid #eee; cursor:pointer;";
                div.innerText = "👤 " + email;
                div.onclick = () => openChat(email);
                area.appendChild(div);
            });
        }

        function openChat(email) {
            currentTarget = email;
            document.getElementById("view-list").style.display = "none";
            document.getElementById("view-detail").style.display = "flex";
            document.getElementById("target-name").innerText = email;
            // 履歴に追加
            let list = JSON.parse(localStorage.getItem("chatHistory") || "[]");
            if(!list.includes(email)) { list.push(email); localStorage.setItem("chatHistory", JSON.stringify(list)); }
            loadMessages();
        }

        async function sendMessage() {
            const text = document.getElementById("chat-input").value;
            if(!text) return;
            document.getElementById("chat-input").value = "";
            await fetch(GAS_URL, { method: "POST", body: JSON.stringify({ mode: "sendMessage", from: myEmail, to: currentTarget, text: text }) });
            loadMessages();
        }

        async function loadMessages() {
    if (!currentTarget) return;
    try {
        const res = await fetch(GAS_URL, { 
            method: "POST", 
            body: JSON.stringify({ mode: "getMessages", user1: myEmail, user2: currentTarget }) 
        });
        const result = await res.json();
        
        if (result.status === "success") {
            const area = document.getElementById("message-area");
            // 今表示されている内容と取得した件数が違う場合のみ書き換える（チラつき防止）
            if (area.children.length !== result.message.length) {
                area.innerHTML = "";
                result.message.forEach(msg => {
                    const div = document.createElement("div");
                    div.className = "bubble " + (msg.from === myEmail ? "msg-right" : "msg-left");
                    div.innerText = msg.text;
                    area.appendChild(div);
                });
                area.scrollTop = area.scrollHeight;
            }
        }
    } catch (e) {
        console.error("受信エラー:", e);
    }
}

        function logout() { localStorage.clear(); window.location.href = "index.html"; }
